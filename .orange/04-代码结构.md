## 关键文件详解

### `libs/core/langchain_core/tools/base.py`

```python
class BaseTool(BaseModel):
    name: str
    description: str
    args_schema: Optional[Type[BaseModel]] = None
    return_direct: bool = False

    # response_format 控制工具返回内容如何处理
    response_format: Literal["content", "content_and_artifact"] = "content"
    # content: 直接作为字符串返回给 LLM
    # content_and_artifact: 返回 (content, artifact) 元组，artifact 不发给 LLM

    # extras: vendor-specific 扩展
    # Anthropic 缓存控制:
    extras: dict = {}  # {"cache_control": {"type": "ephemeral"}}
```

### `libs/langchain/langchain_classic/agents/agent.py`（已废弃）

```python
class AgentExecutor(Chain):
    agent: Union[BaseSingleActionAgent, BaseMultiActionAgent]
    tools: Sequence[BaseTool]
    max_iterations: int = 15  # 最大迭代次数
    max_execution_time: Optional[float] = None  # 超时（秒）
    early_stopping_method: str = "force"  # "force" | "generate"

    def _perform_agent_action(self, agent_action: AgentAction) -> AgentStep:
        tool = next(t for t in self.tools if t.name == agent_action.tool)
        observation = tool.run(agent_action.tool_input)
        return AgentStep(action=agent_action, observation=observation)

    def _construct_scratchpad(self, intermediate_steps):
        """将工具调用历史格式化为 LLM 输入"""
        scratchpad = ""
        for action, observation in intermediate_steps:
            scratchpad += f"\nAction: {action.tool}\nAction Input: {action.tool_input}"
            scratchpad += f"\nObservation: {observation}"
        return scratchpad
```

class AgentMiddleware(Protocol):
    async def before_agent(self, state: AgentState) -> AgentState: ...
    async def after_agent(self, state: AgentState) -> AgentState: ...

## @tool 装饰器实现

```python
# langchain-core/tools/base.py

def tool(
    *args,
    return_direct: bool = False,
    args_schema: Optional[Type[BaseModel]] = None,
    response_format: str = "content",
):
    """@tool 装饰器是 StructuredTool.from_function() 的语法糖"""
    def decorator(func):
        return StructuredTool.from_function(
            func=func,
            name=func.__name__,
            description=func.__doc__,
            return_direct=return_direct,
            args_schema=args_schema or _infer_args_schema(func),
            response_format=response_format,
        )
    return decorator
```
