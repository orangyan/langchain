# 14 如何评测

## 14.1 LangChain 的测试生态

LangChain 作为框架，测试分三层：
1. **单元测试**：测试工具、中间件、LCEL 管道
2. **集成测试**：测试 LLM 集成、工具集成（需要 API Key）
3. **端到端测试**：测试完整 Agent 行为

## 14.2 工具单元测试

```python
# tests/unit_tests/test_tools.py
import pytest
from langchain_core.tools import tool

@tool
def calculator(expression: str) -> str:
    """Calculate a math expression."""
    return str(eval(expression))

def test_tool_name_and_schema():
    """测试工具的元数据"""
    assert calculator.name == "calculator"
    assert "expression" in calculator.args
    assert calculator.description is not None

def test_tool_execution():
    """测试工具实际执行"""
    result = calculator.invoke({"expression": "2 + 2"})
    assert result == "4"

def test_tool_returns_string():
    """测试工具返回类型"""
    result = calculator.invoke({"expression": "3 * 7"})
    assert isinstance(result, str)

def test_tool_schema_validation():
    """测试输入验证"""
    with pytest.raises(Exception):
        calculator.invoke({})  # 缺少必要参数
```

## 14.3 中间件单元测试

```python
# tests/unit_tests/test_middleware.py
import pytest
from unittest.mock import AsyncMock, MagicMock
from langchain.agents.middleware import model_call_limit
from langchain.agents.middleware.types import ModelRequest, AgentState

@pytest.fixture
def mock_request():
    return ModelRequest(
        messages=[],
        tools=[],
        model_kwargs={}
    )

@pytest.mark.asyncio
async def test_model_call_limit_allows_within_limit(mock_request):
    """在限制内允许调用"""
    middleware = model_call_limit(max_calls=5)
    for _ in range(5):
        result = await middleware.before_model(mock_request)
        assert result is not None

@pytest.mark.asyncio
async def test_model_call_limit_blocks_exceeding_limit(mock_request):
    """超出限制时抛出异常"""
    middleware = model_call_limit(max_calls=2)
    await middleware.before_model(mock_request)
    await middleware.before_model(mock_request)
    with pytest.raises(Exception):  # ModelCallLimitExceeded
        await middleware.before_model(mock_request)

@pytest.mark.asyncio
async def test_logging_middleware():
    """测试自定义日志中间件"""
    logs = []

    class LoggingMiddleware:
        async def before_model(self, request):
            logs.append(f"before_model: {len(request.messages)} messages")
            return request
        async def after_model(self, state):
            logs.append("after_model")
            return state

    middleware = LoggingMiddleware()
    await middleware.before_model(ModelRequest(messages=[], tools=[], model_kwargs={}))
    assert len(logs) == 1
    assert "0 messages" in logs[0]
```

## 14.4 使用 Mock LLM 测试 Agent

```python
# tests/unit_tests/test_agent.py
from langchain_core.messages import AIMessage, HumanMessage, ToolMessage
from langchain_core.language_models.fake import FakeMessagesListChatModel

def test_agent_tool_call_flow():
    """测试 Agent 工具调用完整流程（无真实 LLM）"""
    from langchain_core.tools import tool
    from langchain.agents import create_agent
    import asyncio

    @tool
    def get_weather(city: str) -> str:
        """Get weather for a city."""
        return f"Sunny in {city}"

    # Mock LLM：第一次返回工具调用，第二次返回最终答案
    mock_llm = FakeMessagesListChatModel(responses=[
        AIMessage(
            content="",
            tool_calls=[{
                "id": "call_1",
                "name": "get_weather",
                "args": {"city": "Paris"}
            }]
        ),
        AIMessage(content="The weather in Paris is sunny.")
    ])

    agent = create_agent(model=mock_llm, tools=[get_weather])
    result = asyncio.run(
        agent.ainvoke({"messages": [HumanMessage("What's the weather in Paris?")]})
    )

    messages = result["messages"]
    # 验证流程：Human → AI(tool_call) → Tool → AI(final)
    assert len(messages) == 4
    assert isinstance(messages[1], AIMessage)
    assert len(messages[1].tool_calls) == 1
    assert isinstance(messages[2], ToolMessage)
    assert "Sunny" in messages[2].content
    assert "sunny" in messages[3].content.lower()
```

## 14.5 LCEL 管道测试

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.language_models.fake import FakeListChatModel

def test_lcel_pipeline():
    """测试 LCEL 管道（无真实 LLM）"""
    mock_llm = FakeListChatModel(responses=["Paris is the capital of France."])

    chain = (
        ChatPromptTemplate.from_template("Answer: {question}")
        | mock_llm
        | StrOutputParser()
    )

    result = chain.invoke({"question": "Capital of France?"})
    assert isinstance(result, str)
    assert "Paris" in result

@pytest.mark.asyncio
async def test_lcel_streaming():
    """测试 LCEL 流式输出"""
    mock_llm = FakeListChatModel(responses=["Hello World"])
    chain = mock_llm | StrOutputParser()

    chunks = []
    async for chunk in chain.astream("test"):
        chunks.append(chunk)

    assert len(chunks) > 0
    assert "".join(chunks) == "Hello World"
```

## 14.6 LangSmith 评测集成

```python
from langsmith import Client
from langsmith.evaluation import evaluate

client = Client()

# 创建数据集
dataset = client.create_dataset("capital_questions")
client.create_examples(
    inputs=[
        {"question": "Capital of France?"},
        {"question": "Capital of Japan?"},
    ],
    outputs=[
        {"answer": "Paris"},
        {"answer": "Tokyo"},
    ],
    dataset_id=dataset.id,
)

# 定义评测函数
def exact_match(run, example):
    prediction = run.outputs.get("output", "")
    expected = example.outputs.get("answer", "")
    return {"score": int(expected.lower() in prediction.lower())}

# 运行评测
results = evaluate(
    lambda inputs: chain.invoke(inputs),
    data=dataset.name,
    evaluators=[exact_match],
    experiment_prefix="capital-test",
)
```

## 14.8 与其他项目评测对比

| 特性 | LangChain | LangGraph | deer-flow | hermes-agent | nanobot | gstack |
|------|----------|-----------|-----------|-------------|---------|--------|
| **测试框架** | pytest | pytest | pytest | 无 | pytest | bun test |
| **Mock LLM** | FakeListChatModel | FakeListChatModel | FakeListChatModel | N/A | FakeListChatModel | N/A |
| **评测平台** | LangSmith | LangSmith | LangSmith | 无 | 无 | LLM judge |
| **标准测试套件** | ✅ standard-tests | ❌ | ❌ | ❌ | ❌ | ❌ |
| **集成测试** | ✅ (需 API Key) | ✅ | ✅ | ❌ | ✅ | ✅ (E2E) |
| **成本控制** | FakeListChatModel | FakeListChatModel | FakeListChatModel | N/A | FakeListChatModel | 三级测试金字塔 |
| **CI/CD 集成** | ✅ GitHub Actions | ✅ | ✅ | ❌ | ❌ | ✅ |
| **评测数据集** | LangSmith 数据集 | 无 | 无 | 无 | 无 | planted-bug |

**关键洞察：**
- `FakeListChatModel` / `FakeMessagesListChatModel` 是无 API 成本测试的关键
- `langchain/standard-tests` 是 LangChain 独有的 Partner 集成测试规范
- LangSmith 提供完整的追踪、评测、数据集管理闭环
- 与 gstack 的三级成本金字塔思路相同：单元测试 → 集成测试 → 端到端测试
